mkdir $SCRATCH/SORRM_WMT_DVD
git clone -b sorrm_w_dvd git@github.com:dylanschlichting/E3SM.git

cd SORRM_WMT_DVD/E3SM

# to avoid needing to enter your password multiple times:
    git config --global credential.helpter store
    eval "$(ssh-agent -s)"
    ssh-add ~/.ssh/id_ed25519 
    # or whatever ssh id you want to add
    # then enter your password when prompted


git submodule update --init --recursive

export E3SM_REPO=/pscratch/sd/k/kuyeda/SORRM_WMT_DVD/E3SM
cd $E3SM_REPO/cime/scripts

# define path to where you want the case to be built
export E3SM_CASE=/pscratch/sd/k/kuyeda/e3sm_scratch/pm-cpu/QU240_GMPASO-JRA-DVD_tests

# Create case
./create_newcase \
  -case $E3SM_CASE \
  -compiler gnu \
  -mach pm-cpu \
  -project m5260 \
  -compset GMPAS-IAF \
  -res T62_oQU240

# go into case directory & change processor + run time settings
cd $E3SM_CASE

# Processor settings, 128 cores
./xmlchange NTASKS_ATM=128
./xmlchange NTASKS_CPL=128
./xmlchange NTASKS_OCN=128
./xmlchange NTASKS_WAV=128
./xmlchange NTASKS_GLC=128
./xmlchange NTASKS_ICE=128
./xmlchange NTASKS_ROF=128
./xmlchange NTASKS_LND=128
./xmlchange NTASKS_ESP=128
./xmlchange NTASKS_IAC=128

# Let's do a 6 month run. This will only take a few minutes of wall time
./xmlchange STOP_OPTION=nmonths
./xmlchange STOP_N=6

# Write a restart file every 6 months
./xmlchange REST_OPTION=nmonths
./xmlchange REST_N=6


# Now, since we changed the processor settings, we will recursively set up the case otherwise the model won't know that happened
# Note if we didn't change processor settings, a simple ./case.setup would work
./case.setup --r


# Compile the model, this will take a few minutes to up to 10 minutes depending on number of active components.
./case.build

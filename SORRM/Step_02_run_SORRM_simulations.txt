# Once it's done, there are a few things we will change to run with DVD and WMT. We need snapshots of layer thickness to get dM/dt. 
# Editing streams and namelists are different than standalone MPAS. 
# There will be a lot of stuff in /run, and the runtime namelists will be named differently for each component in this folder.

# To edit streams:

cp run/streams.ocean SourceMods/src.mpaso/.

# SourceMods/src.mpaso/streams.ocean will overwrite anything in the run directory. 
# In most coupled simulations, MPAS-O has snapshots turned off by default. Do:

vi SourceMods/src.mpaso/streams.ocean

# Then edit the output interval to turn on snapshots output 

<stream name="output"
        type="output"
        io_type="pnetcdf,cdf5"
        filename_template="QU240_CMPASO-JRA-DVD_tests.mpaso.hist.$Y-$M-$D_$S.nc"
        filename_interval="00-01-00_00:00:00"
        clobber_mode="truncate"
        reference_time="01-01-01_00:00:00"
        output_interval="00-01-00_00:00:00">

# To change the mpaso namelist, edit user_nl_mpaso, which overrides mpaso_in do:
vi user_nl_mpaso

# then add the following
config_am_timeseriesstatsmonthly_enable = .true.
config_use_numericalmixingtracers = .true.
config_AM_discreteVarianceDecay_enable = .true.
config_AM_discreteVarianceDecay_compute_interval = 'dt'
config_AM_discreteVarianceDecay_output_stream = 'discreteVarianceDecayOutput'
config_AM_discreteVarianceDecay_compute_on_startup = .false.
config_AM_discreteVarianceDecay_write_on_startup = .false.

# Processor settings, 128 cores
./xmlchange CPL_NTASKS=1920
./xmlchange CPL_ROOTPE=0

./xmlchange ATM_NTASKS=1920
./xmlchange ATM_ROOTPE=0

./xmlchange LND_NTASKS=1920
./xmlchange LND_ROOTPE=0

./xmlchange ICE_ROOTPE=0

./xmlchange WAV_ROOTPE=0

./xmlchange IAC_NTASKS=1
./xmlchange IAC_ROOTPE=0

./xmlchange ESP_NTASKS=1
./xmlchange ESP_ROOTPE=0

# change the run time directory
#First, change the run directory
./xmlchange RUNDIR=/global/cfs/cdirs/m5260/kuyeda/20260318.v3.SORRME3r3.CRYO2010.bluepulse-beta1-dvd.perlmutter/run

# Now change the build directory
./xmlchange EXEROOT=/global/cfs/cdirs/m5260/kuyeda/20260318.v3.SORRME3r3.CRYO2010.bluepulse-beta1-dvd.perlmutter/bld

# runtime settings
# Tell the model to run in restart mode
./xmlchange RUN_TYPE=hybrid
./xmlchange RUN_REFCASE=20260520.v3.SORRME3r3.CRYO2010.bluepulse-control.chrysalis

# Point the model to Carolyn's restart files -- change this to where you have the files stored

./xmlchange GET_REFCASE=TRUE

# Complete simulation after 1 month! 
./xmlchange STOP_OPTION=nmonths
./xmlchange STOP_N=1

#Change the reference date to year 200
./xmlchange RUN_REFDATE=0200-01-01

# Change the requested wall time to 2 hours
./xmlchange JOB_WALLCLOCK_TIME=02:00:00

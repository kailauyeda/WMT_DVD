cd /pscratch/sd/k/kuyeda/e3sm_scratch/pm-cpu/QU240_GMPASO-JRA-DVD_tests

# Replace with the path to your sorrm branch
export E3SM_REPO=/pscratch/sd/k/kuyeda/SORRM_WMT_DVD/E3SM

# Now, go into the scripts repo 
cd $E3SM_REPO/cime/scripts

# Define the path to where you want to the code to be built
export E3SM_CASE=/pscratch/sd/k/kuyeda/e3sm_scratch/pm-cpu/QU240_GMPASO-JRA_DVD_tests_test

# Note -- change to Henri's project
./create_newcase \
  -case $E3SM_CASE \
  -compiler gnu \
  -mach pm-cpu \
  -project m5260 \
  -compset GMPAS-IAF \
  -res T62_oQU240

cd $E3SM_CASE

module load python

# Processor settings, 128 cores
./xmlchange NTASKS_ATM=128
./xmlchange NTASKS_CPL=128
./xmlchange NTASKS_OCN=128
./xmlchange NTASKS_WAV=128
./xmlchange NTASKS_GLC=128
./xmlchange NTASKS_ICE=128
./xmlchange NTASKS_ROF=128
./xmlchange NTASKS_LND=128
./xmlchange NTASKS_ESP=128
./xmlchange NTASKS_IAC=128

# Let's do a 6 month run. This will only take a few minutes of wall time
./xmlchange STOP_OPTION=nmonths
./xmlchange STOP_N=6
# Wrtie a restart file every 6 months
./xmlchange REST_OPTION=nmonths
./xmlchange REST_N=6

# Note if we didn't change processor settings, a simple ./case.setup would work
./case.setup --r

./case.build
